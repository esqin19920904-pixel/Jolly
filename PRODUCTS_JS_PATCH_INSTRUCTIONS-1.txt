    let list = _filteredAll;
    if (q) {
      list = list.filter(p => {
        const hay = [p.name, p.mainCode, p.extraCodeValue, p.color, p.note, ...(p.barcodes || [])].filter(Boolean).join(' ').toLowerCase();
